% ============================================================
% Latexdiff custom preamble for Yellow Marker Highlighting
% ============================================================
\RequirePackage{color,soul}
\RequirePackage[normalem]{ulem}

\sethlcolor{yellow}

% Register standard citation and cross-reference commands for soul
\soulregister\cite7
\soulregister\citet7
\soulregister\citep7
\soulregister\citeauthor7
\soulregister\citeyear7
\soulregister\ref7
\soulregister\eqref7
\soulregister\pageref7
\soulregister\autoref7
\soulregister\url7

% Toggle switch: set to false to suppress markup and render clean text
\newif\ifshowdiffs
\showdiffstrue

\ifshowdiffs
  \DeclareRobustCommand{\DIFadd}[1]{{\protect\hl{#1}}}
  \DeclareRobustCommand{\DIFdel}[1]{{\color{red}\sout{#1}}}
\else
  \DeclareRobustCommand{\DIFadd}[1]{#1}
  \DeclareRobustCommand{\DIFdel}[1]{}
\fi

\providecommand{\DIFaddbegin}{}
\providecommand{\DIFaddend}{}
\providecommand{\DIFdelbegin}{}
\providecommand{\DIFdelend}{}
\providecommand{\DIFaddFL}[1]{\DIFadd{#1}}
\providecommand{\DIFdelFL}[1]{\DIFdel{#1}}
\providecommand{\DIFaddbeginFL}{}
\providecommand{\DIFaddendFL}{}
\providecommand{\DIFdelbeginFL}{}
\providecommand{\DIFdelendFL}{}

% Protect PDF bookmarks and outlines
\makeatletter
\AtBeginDocument{%
  \pdfstringdefDisableCommands{%
    \def\DIFadd#1{#1}%
    \def\DIFdel#1{}%
    \def\hl#1{#1}%
  }%
}
\makeatother
